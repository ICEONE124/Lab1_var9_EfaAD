typedef enum {
    TYPE_SCALAR,
    TYPE_ROW_VEC,
    TYPE_COL_VEC,
    TYPE_MATRIX
} DataType;

typedef struct {
    DataType type;
    int rows;
    int cols;
    double *data;
} DataContainer;

typedef struct Item {
    struct Item *prev;
    struct Item *next;
} Item;

typedef struct List {
    Item *head;
    Item *tail;
} List;
